    # symbol

    //demo1
let s1 = Symbol()
let s2 = Symbol()
s1 == s2//false

//demo2 
  ({a:1} == {a:1})//false

//symbol 可以作为对象的key
const obj = {};
const sym = Symbol();
obj[sym] = 'foo';
obj.bar = 'bar';
console.log(obj); // { bar: 'bar' }
console.log(sym in obj); // true
console.log(obj[sym]); // foo
console.log(Object.keys(obj)); // ['bar'],获取不到symbol
console.log(Reflect.ownKeys(obj)); // ['bar','Symbol()]
